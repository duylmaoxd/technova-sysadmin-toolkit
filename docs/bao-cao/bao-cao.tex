\chapter{Giới thiệu}

\section{Bài toán}
Công ty giả định \textbf{TechNova Solutions} có một máy chủ Ubuntu tập trung (\cmd{technova-core-srv}); lập trình viên, data analyst và tester SSH vào làm việc. \textbf{Anh Hưng} — Tech Lead — làm việc từ xa, chỉ dùng terminal qua SSH, cần:
\begin{enumerate}
  \item cấp quyền và quản lý nhân viên, kể cả khi nghỉ việc;
  \item biết ai đang làm gì, ai treo process ngốn CPU/RAM;
  \item nhận cảnh báo tức thời khi có sự cố hoặc bị tấn công, kể cả ban đêm;
  \item tự động bảo dưỡng và sao lưu, có bản sao nằm ngoài máy chủ.
\end{enumerate}
Làm tay mỗi việc gồm hàng chục lệnh rời rạc, dễ sai và khó lặp lại. Đồ án \textbf{tự động hoá toàn bộ bằng Bash}.

\section{Ứng dụng thực tế}
Phù hợp doanh nghiệp nhỏ có vài máy chủ Linux nhưng không có đội IT riêng để vận hành Zabbix, Wazuh hay Veeam. Toolkit xử lý các rủi ro thường gặp: nhân viên nghỉ việc vẫn còn quyền truy cập; bot dò mật khẩu SSH; ổ đầy hoặc process treo lúc nửa đêm; mất dữ liệu do hỏng ổ hay mã độc tống tiền. Toolkit chỉ cần Bash và lệnh có sẵn của Linux, cài bằng một lệnh, giao diện tiếng Việt.

\section{Kết quả đạt được}
\begin{itemize}
  \item Khoảng \textbf{4.370 dòng Bash}, \textbf{71 lệnh}, chia thành \textbf{5 chức năng}: \cmd{users}, \cmd{monitor}, \cmd{alert}, \cmd{backup}, \cmd{scheduler}.
  \item Hai chế độ dùng qua SSH: giao diện menu (TUI) và dòng lệnh (CLI).
  \item Cài \textbf{trực tiếp lên máy Ubuntu thật} và tự vận hành bằng systemd timer; cảnh báo qua Telegram/email; sao lưu lên Google Drive; phát hiện và chặn tấn công SSH.
  \item Kiểm thử tự động \textbf{75/75 PASS}; shellcheck không còn cảnh báo.
\end{itemize}
Ngoài đề tài 3, toolkit thực hành nội dung của các đề tài 1 (tiến trình), 4 (systemd), 5 (user \& bảo mật), 6 (SSH, tường lửa), 9 (giám sát, log) và 10 (cron, cảnh báo email).

\chapter{Cơ sở lý thuyết}

\section{Bash scripting}
Bash là shell mặc định của Linux, có biến, điều kiện, vòng lặp, hàm, mảng. Mọi script của toolkit bắt đầu bằng chế độ ``nghiêm ngặt'':
\begin{Verbatim}
#!/usr/bin/env bash
set -Eeuo pipefail     # dừng khi lỗi, cấm biến chưa khai báo, bắt lỗi trong pipeline
trap 'on_error $LINENO' ERR
\end{Verbatim}
Các kỹ thuật chính: hàm và mã thoát, \cmd{getopts}, mảng kết hợp, biểu thức chính quy, \cmd{awk}/\cmd{sed} để xử lý log, \cmd{trap}, \cmd{flock} chống chạy trùng.

\section{Quản trị Linux liên quan}
\begin{itemize}
  \item \textbf{User và quyền:} \cmd{/etc/passwd}, \cmd{/etc/shadow}, \cmd{/etc/group}; quyền \textit{rwx} dạng bát phân; \textbf{SGID} (tệp mới kế thừa nhóm của thư mục) và \textbf{sticky bit} (chỉ xoá được tệp của mình); sudo; \cmd{chage} cho chính sách mật khẩu. Tài khoản có UID 0 có toàn quyền root.
  \item \textbf{Tiến trình:} \cmd{ps}, \cmd{top}, \cmd{kill} (SIGTERM, SIGKILL); bộ lập lịch CFS chia CPU theo giá trị \textbf{nice} ($-20$ đến $19$).
  \item \textbf{systemd:} PID 1 sau khi kernel khởi động; quản lý \textit{unit}. Một \cmd{.service} mô tả tác vụ, một \cmd{.timer} kích hoạt service theo lịch; log được \textbf{journald} thu thập (\cmd{journalctl -u}).
  \item \textbf{cron / at:} cron chạy định kỳ theo 5 trường \textit{phút giờ ngày tháng thứ}; \cmd{at} hẹn chạy một lần.
  \item \textbf{Nhật ký:} journald và rsyslog (\cmd{/var/log/auth.log}). Mỗi lần đăng nhập SSH sai để lại dòng như \cmd{Failed password ... from <IP>} hoặc, khi máy chỉ cho đăng nhập bằng key, \cmd{Invalid user ... [preauth]}.
  \item \textbf{Mạng:} SSH cổng 22; tường lửa netfilter điều khiển bằng \cmd{iptables} hoặc \cmd{ufw}.
\end{itemize}

\section{Sao lưu}
Nén \cmd{tar}, kiểm tra toàn vẹn bằng \textbf{SHA-256}, xoay vòng giữ $N$ bản, và theo \textbf{quy tắc 3-2-1}: luôn có một bản ở nơi khác. Toolkit đẩy bản sao lên \textbf{Google Drive} bằng \cmd{rclone} (xác thực OAuth, chỉ xin quyền \cmd{drive.file} — chỉ thấy tệp do nó tạo).

\section{Hệ thống phát hiện xâm nhập (IDS)}
IDS là hệ thống theo dõi để phát hiện dấu hiệu tấn công và cảnh báo. Có hai cách phân loại:
\begin{itemize}
  \item theo vị trí: \textbf{NIDS} soi gói tin mạng (Snort, Suricata); \textbf{HIDS} theo dõi chính máy chủ qua log và tệp (OSSEC, Wazuh, fail2ban, AIDE);
  \item theo cách phát hiện: \textbf{dựa trên luật} — so với mẫu tấn công đã biết, kèm ngưỡng; \textbf{dựa trên bất thường} — học trạng thái bình thường bằng thống kê hoặc học máy.
\end{itemize}
Phần lớn IDS dùng thực tế là dựa trên luật vì dễ giải thích và ít báo nhầm. IDS của toolkit là \textbf{HIDS dựa trên luật} kết hợp \textbf{kiểm tra toàn vẹn tệp} (giống fail2ban + AIDE), \textbf{không dùng học máy}.

\cmd{users create} tạo nhân viên từ \cmd{users.csv}: nhóm phòng ban, mật khẩu ngẫu nhiên, bắt đổi mật khẩu ở lần đăng nhập đầu. \cmd{users setup-dirs} tạo thư mục phòng ban quyền \cmd{2770} (SGID) và thư mục chung \cmd{1777} (sticky). \cmd{users offboard} xử lý nghỉ việc trong một lệnh: khoá tài khoản, ngắt phiên, thu sudo, xoá crontab, lưu trữ home.
\anhdemo{03-phan-quyen.png}{Nhân viên phòng data không vào được thư mục phòng dev}{Laptop 2 đăng nhập \texttt{cuonglv}: \texttt{ls /srv/technova/dev} bị từ chối; \texttt{ls -l /srv/technova/data}.}
\anhdemo{04-offboard.png}{Nhân viên nghỉ việc: khoá và ngắt phiên ngay}{\texttt{sudo sysadmin users offboard binhtt}.}

\section{monitor — giám sát}
\cmd{monitor who} cho biết ai đang đăng nhập và từ đâu; \cmd{monitor usage} cộng CPU/RAM theo từng nhân viên; \cmd{monitor hogs} tìm process vượt ngưỡng; xử lý bằng \cmd{renice} hoặc \cmd{kill}. \cmd{monitor logcheck} tổng hợp log: SSH thất bại theo IP, đăng nhập thành công, sudo, thay đổi tài khoản, dịch vụ lỗi, lỗi kernel.
\anhdemo{05-monitor-hogs.png}{Phát hiện nhân viên treo process ngốn CPU}{Laptop 2 chạy process nặng; Laptop 1: \texttt{sysadmin monitor usage} và \texttt{sysadmin monitor hogs}.}

\section{alert — cảnh báo và phát hiện tấn công}
\cmd{alert check} đo CPU, RAM, ổ đĩa, process treo, dịch vụ lỗi và gọi IDS; có sự cố thì gửi Telegram/email, hết sự cố thì gửi tin ``đã ổn định''.

\section{backup — sao lưu lên Google Drive}
\cmd{backup run}: nén \cmd{tar.gz} $\rightarrow$ kiểm tra $\rightarrow$ SHA-256 $\rightarrow$ giữ 7 bản $\rightarrow$ tải lên Google Drive (xoá bản cũ hơn 30 ngày) $\rightarrow$ báo lỗi nếu thất bại. Khôi phục bằng \cmd{backup restore}.
\anhdemo{11-backup-run.png}{Sao lưu và tải lên Google Drive}{\texttt{sudo sysadmin backup run}.}

\section{scheduler — systemd}
Toolkit tự tạo các unit trong \cmd{/etc/systemd/system/}; ví dụ dịch vụ sao lưu:
\begin{Verbatim}
[Service]
Type=oneshot
ExecStart=-/opt/sysadmin-toolkit/sysadmin.sh --yes backup run
ExecStart=/opt/sysadmin-toolkit/sysadmin.sh --yes monitor log-upload
\end{Verbatim}
\anhdemo{13-journalctl-backup.png}{Nhật ký các đêm sao lưu trong journald}{\texttt{journalctl -u toolkit-backup --since "3 days ago" | tail -20}.}

# Cài đặt trên máy Ubuntu
